cmake_minimum_required(VERSION 3.22)
project(mosaico_preview C ASM)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(LV_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/managed_components/lvgl__lvgl" lvgl)
set(EMBED "")
if(APPLE)
  string(APPEND EMBED ".section __TEXT,__const\n")
else()
  string(APPEND EMBED ".section .rodata\n")
endif()
foreach(NAME assets font16 font20 font28)
  list(APPEND EMBED_DEPENDS "${ROOT}/main/generated/${NAME}.bin")
  string(APPEND EMBED ".balign 4\n.globl _binary_${NAME}_bin_start\n.globl _binary_${NAME}_bin_end\n_binary_${NAME}_bin_start:\n.incbin \"${ROOT}/main/generated/${NAME}.bin\"\n_binary_${NAME}_bin_end:\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/embed.S" "${EMBED}")
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/embed.S" PROPERTIES OBJECT_DEPENDS "${EMBED_DEPENDS}")
foreach(TARGET preview idle_integration)
  add_executable(${TARGET} ${TARGET}.c native_panel.c "${ROOT}/main/ui.c" "${ROOT}/main/device.c" "${ROOT}/main/schedule.c" "${ROOT}/main/idle_screen.c" "${ROOT}/main/generated/assets.c" "${ROOT}/components/cjson/cJSON.c" "${CMAKE_CURRENT_BINARY_DIR}/embed.S")
  target_include_directories(${TARGET} PRIVATE . "${ROOT}/main" "${ROOT}/components/cjson")
  target_link_libraries(${TARGET} PRIVATE lvgl m)
endforeach()
enable_testing()
foreach(CASE normal missing-touch panel-off-failure panel-on-failure awake-stall wake-stall)
  add_test(NAME idle-${CASE} COMMAND idle_integration ${CASE})
  set_tests_properties(idle-${CASE} PROPERTIES TIMEOUT 30)
endforeach()
